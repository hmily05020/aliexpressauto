  ipcMain.handle("chrome:setLogisticsAttribute", async () => {
    const { tabs } = await getConnectedChromeTabs();
    const editTab = await pickBestDxmEditTab(tabs);
    if (!editTab) throw new Error("没有找到店小秘编辑页");
    await waitForDxmEditContent(editTab);
    const expression = `(() => new Promise(async (resolve) => {
      try {
        const n=(v)=>(v||"").replace(/\\s+/g," ").trim();
        const slp=ms=>new Promise(d=>setTimeout(d,ms));
        const vis=(el)=>{if(!el||!el.getBoundingClientRect)return false;const r=el.getBoundingClientRect();return r.width>0&&r.height>0};
        const clk=(el)=>{el.scrollIntoView({block:"center"});const r=el.getBoundingClientRect();["mouseover","mousedown","mouseup","click"].forEach(t=>el.dispatchEvent(new MouseEvent(t,{bubbles:true,cancelable:true,clientX:r.left+r.width/2,clientY:r.top+r.height/2})))};
        const cards=Array.from(document.querySelectorAll(".form-card")).filter(vis);
        const prodCard=cards.find(c=>n(c.querySelector("h4")?.innerText||"").includes("产品信息"));
        if(!prodCard){resolve({ok:false,reason:"未找到产品信息"});return}
        // Find logistics span.link and click it
        const spans=Array.from(prodCard.querySelectorAll("span.link"));
        const logSpan=spans.find(s=>n(s.innerText||s.textContent)==="批量"&&vis(s));
        if(!logSpan){resolve({ok:false,reason:"未找到物流批量按钮"});return}
        document.body.click();await slp(300);
        clk(logSpan);await slp(1200);
        // Force-show modal (Vue may not respond to programmatic click)
        const modal=document.querySelector(".batch-edit-modal");
        if(modal){modal.style.display="";modal.style.removeProperty("display");await slp(500)}
        if(!modal||(modal.getAttribute("style")||"").indexOf("display: none")>=0){resolve({ok:false,reason:"弹窗未打开"});return}
        // Click 普货 checkbox
        const wrappers=Array.from(modal.querySelectorAll(".ant-checkbox-wrapper"));
        const cargo=wrappers.find(w=>n(w.innerText||w.textContent)==="普货");
        if(!cargo){resolve({ok:false,reason:"未找到普货"});return}
        const cb=cargo.querySelector("input[type=checkbox]")||cargo.querySelector(".ant-checkbox-input");
        const wasChecked=cb?cb.checked:false;
        clk(cb||cargo);await slp(400);
        // Click 确定
        const btns=Array.from(modal.querySelectorAll("button")).filter(vis);
        const ok=btns.find(b=>n(b.innerText||b.textContent)==="确定");
        if(ok){clk(ok);await slp(500)}
        resolve({ok:true,modalFound:!!modal,cargoFound:!!cargo,wasChecked:wasChecked,nowChecked:cb?cb.checked:null,okClicked:!!ok});
      } catch(e) { resolve({ok:false,error:e.message||String(e)}); }
    }))`;
    const result = await evaluateInTab(editTab, expression);
    appendAutomationLog("setLogisticsAttribute", JSON.stringify(result));
    return result;
  });
